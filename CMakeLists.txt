cmake_minimum_required(VERSION 3.21)
project(Aurora86 VERSION 1.6.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
# Keep local builder paths out of distributed binaries.
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 add_compile_options("-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=." "-ffile-prefix-map=${CMAKE_CURRENT_BINARY_DIR}=./build"
                     "-fdebug-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=.")
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Widgets Network Test
             OPTIONAL_COMPONENTS Multimedia WebEngineWidgets)
# WebEngine (Chromium) and Multimedia are optional: without them the station still
# builds, with an offline Network window and a silent Speaker.
set(AURORA_WITH_MULTIMEDIA 0)
set(AURORA_WITH_WEBENGINE 0)
set(AURORA_EXTRA_LIBS "")
find_package(ZLIB QUIET)
set(AURORA_WITH_ZLIB 0)
if(ZLIB_FOUND)
 set(AURORA_WITH_ZLIB 1)
 list(APPEND AURORA_EXTRA_LIBS ZLIB::ZLIB)
endif()
if(TARGET Qt6::Multimedia)
 set(AURORA_WITH_MULTIMEDIA 1)
 list(APPEND AURORA_EXTRA_LIBS Qt6::Multimedia)
endif()
if(TARGET Qt6::WebEngineWidgets)
 set(AURORA_WITH_WEBENGINE 1)
 list(APPEND AURORA_EXTRA_LIBS Qt6::WebEngineWidgets)
endif()
file(GLOB AURORA_LUA_SOURCES CONFIGURE_DEPENDS third_party/lua/*.c)
add_library(aurora_lua STATIC ${AURORA_LUA_SOURCES})
target_include_directories(aurora_lua PUBLIC third_party/lua)
set_target_properties(aurora_lua PROPERTIES POSITION_INDEPENDENT_CODE ON AUTOMOC OFF AUTORCC OFF)
if(WIN32)
 target_compile_definitions(aurora_lua PRIVATE LUA_USE_WINDOWS)
elseif(UNIX)
 target_compile_definitions(aurora_lua PRIVATE LUA_USE_LINUX)
 target_link_libraries(aurora_lua PUBLIC m ${CMAKE_DL_LIBS})
endif()
add_library(aurora_core src/system.cpp src/shell.cpp src/packages.cpp src/lua_runtime.cpp src/mods.cpp src/basic.cpp src/archive.cpp)
target_include_directories(aurora_core PUBLIC src)
target_compile_definitions(aurora_core PUBLIC
 AURORA_WITH_ZLIB=${AURORA_WITH_ZLIB}
 AURORA_WITH_MULTIMEDIA=${AURORA_WITH_MULTIMEDIA}
 AURORA_WITH_WEBENGINE=${AURORA_WITH_WEBENGINE})
target_link_libraries(aurora_core PUBLIC Qt6::Core aurora_lua)
if(MSVC)
 add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/utf-8>")
 target_compile_options(aurora_core PRIVATE /W4 /utf-8)
else()
 target_compile_options(aurora_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
find_package(Threads REQUIRED)
add_library(aurora_doom_engine STATIC src/doom_engine.c)
set_target_properties(aurora_doom_engine PROPERTIES AUTOMOC OFF AUTORCC OFF)
if(UNIX)
 target_link_libraries(aurora_doom_engine PUBLIC m)
endif()
add_executable(aurora86-doom src/doom_worker.cpp)
target_link_libraries(aurora86-doom PRIVATE aurora_doom_engine Qt6::Core Threads::Threads)
set(AURORA_UI_SOURCES src/lua_apps.cpp src/login.cpp src/display.cpp src/package_apps.cpp src/desktop.cpp src/apps.cpp src/games.cpp src/creative_games.cpp src/wallpaper.cpp src/audio.cpp src/screensaver.cpp src/hellgate.cpp src/doom.cpp assets/resources.qrc assets/doom.qrc)
qt_add_executable(aurora86 WIN32 src/main.cpp ${AURORA_UI_SOURCES})
add_dependencies(aurora86 aurora86-doom)
if(WIN32)
 target_sources(aurora86 PRIVATE packaging/windows/aurora86.rc)
 target_include_directories(aurora86 PRIVATE packaging/windows)
endif()
target_link_libraries(aurora86 PRIVATE aurora_core Qt6::Widgets Qt6::Network ${AURORA_EXTRA_LIBS})
if(MSVC)
 target_compile_options(aurora86 PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/utf-8>")
else()
 target_compile_options(aurora86 PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(doom_test tests/doom_test.cpp)
 target_link_libraries(doom_test PRIVATE Qt6::Core Qt6::Test)
 target_compile_definitions(doom_test PRIVATE AURORA_DOOM_TEST_WAD="${CMAKE_CURRENT_SOURCE_DIR}/third_party/doom/freedoom1.wad")
 add_dependencies(doom_test aurora86-doom)
 add_test(NAME doom COMMAND doom_test)
 set_tests_properties(doom PROPERTIES TIMEOUT 20)
 add_executable(core_test tests/core_test.cpp)
 target_link_libraries(core_test PRIVATE aurora_core Qt6::Test)
 add_executable(ui_test tests/ui_test.cpp ${AURORA_UI_SOURCES})
 target_link_libraries(ui_test PRIVATE aurora_core Qt6::Widgets Qt6::Network ${AURORA_EXTRA_LIBS} Qt6::Test)
 add_test(NAME ui COMMAND ui_test)
 set_tests_properties(ui PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;AURORA_NO_AUDIO=1")
 add_executable(lua_test tests/lua_test.cpp)
 target_link_libraries(lua_test PRIVATE aurora_core Qt6::Test)
 add_test(NAME lua COMMAND lua_test)
 set_tests_properties(lua PROPERTIES TIMEOUT 15)
 add_executable(login_test tests/login_test.cpp src/login.cpp)
 target_link_libraries(login_test PRIVATE aurora_core Qt6::Widgets Qt6::Network Qt6::Test)
 add_test(NAME login COMMAND login_test)
 set_tests_properties(login PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;AURORA_NO_AUDIO=1")
 add_executable(package_test tests/package_test.cpp)
 target_link_libraries(package_test PRIVATE aurora_core Qt6::Test)
 add_test(NAME packages COMMAND package_test)
 add_executable(basic_test tests/basic_test.cpp)
 target_link_libraries(basic_test PRIVATE aurora_core Qt6::Test)
 add_test(NAME basic COMMAND basic_test)
 set_tests_properties(basic PROPERTIES TIMEOUT 30)
 add_executable(shell_test tests/shell_test.cpp)
 target_link_libraries(shell_test PRIVATE aurora_core Qt6::Test)
 add_test(NAME shell COMMAND shell_test)
 set_tests_properties(shell PROPERTIES TIMEOUT 30)
 add_executable(hellgate_test tests/hellgate_test.cpp src/hellgate.cpp)
 target_link_libraries(hellgate_test PRIVATE aurora_core Qt6::Widgets Qt6::Test)
 add_test(NAME hellgate COMMAND hellgate_test)
 set_tests_properties(hellgate PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;AURORA_NO_AUDIO=1")
 add_test(NAME core COMMAND core_test)
 add_test(NAME desktop COMMAND aurora86 --self-test)
 set_tests_properties(desktop PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS aurora86 aurora86-doom RUNTIME DESTINATION bin)

install(FILES packaging/io.github.fis4med2.Aurora86.desktop DESTINATION share/applications)
install(FILES packaging/io.github.fis4med2.Aurora86.metainfo.xml DESTINATION share/metainfo)
install(FILES packaging/aurora86.svg DESTINATION share/icons/hicolor/scalable/apps RENAME io.github.fis4med2.Aurora86.svg)
