cmake_minimum_required(VERSION 3.21)
project(Aurora86 VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Widgets Multimedia WebEngineWidgets Test)
add_library(aurora_core src/system.cpp src/shell.cpp)
target_include_directories(aurora_core PUBLIC src)
target_link_libraries(aurora_core PUBLIC Qt6::Core)
target_compile_options(aurora_core PRIVATE -Wall -Wextra -Wpedantic)
set(AURORA_UI_SOURCES src/desktop.cpp src/apps.cpp src/games.cpp src/creative_games.cpp src/wallpaper.cpp src/audio.cpp assets/resources.qrc)
add_executable(aurora86 src/main.cpp ${AURORA_UI_SOURCES})
target_link_libraries(aurora86 PRIVATE aurora_core Qt6::Widgets Qt6::Multimedia Qt6::WebEngineWidgets)
target_compile_options(aurora86 PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
 add_executable(core_test tests/core_test.cpp)
 target_link_libraries(core_test PRIVATE aurora_core Qt6::Test)
 add_executable(ui_test tests/ui_test.cpp ${AURORA_UI_SOURCES})
 target_link_libraries(ui_test PRIVATE aurora_core Qt6::Widgets Qt6::Multimedia Qt6::WebEngineWidgets Qt6::Test)
 add_test(NAME ui COMMAND ui_test)
 set_tests_properties(ui PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;AURORA_NO_AUDIO=1")
 add_test(NAME core COMMAND core_test)
 add_test(NAME desktop COMMAND aurora86 --self-test)
 set_tests_properties(desktop PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS aurora86 RUNTIME DESTINATION bin)

install(FILES packaging/io.github.fis4med2.Aurora86.desktop DESTINATION share/applications)
install(FILES packaging/io.github.fis4med2.Aurora86.metainfo.xml DESTINATION share/metainfo)
install(FILES packaging/aurora86.svg DESTINATION share/icons/hicolor/scalable/apps RENAME io.github.fis4med2.Aurora86.svg)
